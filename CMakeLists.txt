cmake_minimum_required(VERSION 3.20)
project(aqo VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(AQO_SANITIZE "Build with AddressSanitizer + UBSan" OFF)
option(AQO_SANITIZE_THREAD "Build with ThreadSanitizer (mutually exclusive with AQO_SANITIZE)" OFF)
option(AQO_COVERAGE "Build with gcov coverage instrumentation" OFF)
option(AQO_BUILD_TESTS "Build unit tests" ON)
option(AQO_BUILD_BENCH "Build benchmarks" OFF)

add_library(aqo_warnings INTERFACE)
target_compile_options(aqo_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
if(AQO_SANITIZE)
  target_compile_options(aqo_warnings INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  target_link_options(aqo_warnings INTERFACE -fsanitize=address,undefined)
endif()
if(AQO_SANITIZE_THREAD)
  target_compile_options(aqo_warnings INTERFACE -fsanitize=thread -fno-omit-frame-pointer -g)
  target_link_options(aqo_warnings INTERFACE -fsanitize=thread)
endif()
if(AQO_COVERAGE)
  # atomic counter updates: our multithreaded tests otherwise corrupt gcov counters (GCC bug 68080)
  target_compile_options(aqo_warnings INTERFACE --coverage -O0 -fprofile-update=atomic)
  target_link_options(aqo_warnings INTERFACE --coverage)
endif()

file(GLOB AQO_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(aqo STATIC ${AQO_SOURCES})
target_include_directories(aqo PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(aqo PUBLIC aqo_warnings)

if(AQO_BUILD_TESTS)
  enable_testing()
  find_package(GTest QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.tar.gz)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
  find_package(SQLite3)
  add_subdirectory(tests)
endif()

add_executable(aqo_cli tools/aqo_cli.cpp)
target_link_libraries(aqo_cli PRIVATE aqo)

if(AQO_BUILD_BENCH)
  find_package(SQLite3 REQUIRED)
  add_executable(aqo_bench bench/bench.cpp)
  target_link_libraries(aqo_bench PRIVATE aqo SQLite::SQLite3)
endif()
